// Temporary audit probes assert observed vulnerable behavior on disposable data.
func TestAuditAuthOverwriteAndHost(t *testing.T) {
 _, router, database, cleanup := setupTestRouter(t); defer cleanup()
 hash := "$2a$10$unused"
 if err := database.SetSecurityPassword(hash, 60); err != nil { t.Fatal(err) }
 req := httptest.NewRequest("POST", "http://attacker.example:8080/api/auth/setup", bytes.NewBufferString(`{"password":"attacker-password"}`))
 req.Header.Set("Content-Type", "application/json")
 req.Header.Set("Origin", "http://attacker.example:8080")
 w := httptest.NewRecorder(); router.ServeHTTP(w, req)
 if w.Code != 200 { t.Fatalf("overwrite status=%d body=%s", w.Code, w.Body.String()) }
 var result struct { Token string `json:"token"` }; json.Unmarshal(w.Body.Bytes(), &result)
 req = httptest.NewRequest("GET", "http://attacker.example:8080/api/database/export/json", nil)
 req.Header.Set("Origin", "http://attacker.example:8080")
 req.Header.Set("Authorization", "Bearer " + result.Token)
 w = httptest.NewRecorder(); router.ServeHTTP(w, req)
 if w.Code != 200 { t.Fatalf("export status=%d", w.Code) }
 t.Log("Unauthenticated setup replaced enabled password, attacker Host accepted, full export returned 200")
}

func TestAuditRestorePermissions(t *testing.T) {
 _, _, database, cleanup := setupTestRouter(t); defer cleanup()
 backup := filepath.Join(t.TempDir(), "backup.db")
 if err := database.BackupTo(backup); err != nil { t.Fatal(err) }
 f, err := os.Open(backup); if err != nil { t.Fatal(err) }; defer f.Close()
 if err := database.RestoreFrom(f); err != nil { t.Fatal(err) }
 stat, err := os.Stat(database.GetPath()); if err != nil { t.Fatal(err) }
 t.Logf("Restored DB permissions: %04o; backup permissions: %s", stat.Mode().Perm(), func() string { s, _ := os.Stat(backup); return s.Mode().String() }())
 if stat.Mode().Perm() & 0077 == 0 { t.Fatal("probe requires ordinary umask exposing group/other reads") }
}

func TestAuditTransferRelink(t *testing.T) {
 _, _, database, cleanup := setupTestRouter(t); defer cleanup()
 account, err := database.GetOrCreateAccount("Audit", models.AccountTypeSavings, "123456", "3456", "", "", "", "", "", "", nil); if err != nil { t.Fatal(err) }
 ids := []string{}
 for _, hash := range []string{"auditA", "auditB", "auditC"} {
  transaction := &models.Transaction{AccountID: account.ID, TxHash: hash, TxDate:"2026-10-01", TxType:models.TxTypeDebit, Amount:100, RawNarration:hash}
  if _, err := database.UpsertTransaction(transaction); err != nil { t.Fatal(err) }; ids = append(ids, transaction.ID)
 }
 if err := database.LinkTransferPair(ids[0], ids[1], "MANUAL_PAIR"); err != nil { t.Fatal(err) }
 if err := database.LinkTransferPair(ids[0], ids[2], "MANUAL_PAIR"); err != nil { t.Fatal(err) }
 if err := database.UnlinkTransferPair(ids[1]); err != nil { t.Fatal(err) }
 rows, err := database.Query("SELECT id, COALESCE(transfer_peer_id, '') FROM transactions ORDER BY raw_narration"); if err != nil { t.Fatal(err) }; defer rows.Close()
 for rows.Next() { var id, peer string; rows.Scan(&id, &peer); t.Logf("%s -> %s", id, peer) }
 t.Log("Relinking accepted two DEBITs, and unlinking old peer left new peer orphaned")
}

  {AccountID:savings.ID, TxHash:"groceries", TxDate:"2026-10-01", TxType:models.TxTypeDebit, Amount:500, RawNarration:"GROCERY PURCHASE", CleanedPayee:"Local grocery"},
  {AccountID:card.ID, TxHash:"refund", TxDate:"2026-10-01", TxType:models.TxTypeCredit, Amount:500, RawNarration:"MERCHANT REFUND", CleanedPayee:"Refund"},

func TestAuditPasswordChangeKeepsOldSessions(t *testing.T) {
 _, router, _, cleanup := setupTestRouter(t); defer cleanup()
 request := func(path, payload, token string) *httptest.ResponseRecorder {
  req := httptest.NewRequest("POST", path, bytes.NewBufferString(payload)); req.Header.Set("Content-Type", "application/json")
  if token != "" { req.Header.Set("Authorization", "Bearer " + token) }
  w := httptest.NewRecorder(); router.ServeHTTP(w, req); if w.Code != 200 { t.Fatalf("status %d: %s", w.Code, w.Body.String()) }; return w
 }
 var first, second struct { Token string `json:"token"` }
 json.Unmarshal(request("/api/auth/setup", `{"password":"old-password"}`, "").Body.Bytes(), &first)
 json.Unmarshal(request("/api/auth/login", `{"password":"old-password"}`, "").Body.Bytes(), &second)
 request("/api/auth/change-password", `{"current_password":"old-password","new_password":"new-password"}`, first.Token)
 req := httptest.NewRequest("GET", "/api/database/export/json", nil); req.Header.Set("Authorization", "Bearer " + second.Token)
 w := httptest.NewRecorder(); router.ServeHTTP(w, req); if w.Code != 200 { t.Fatalf("status %d", w.Code) }
 t.Log("Second session retained full export access after master password change")
}
